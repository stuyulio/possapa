        let allRecords = [];

        const STORAGE_KEY = "pos_sapa_2026_dashboard_records_v1";
        const SOURCE_META_KEY = "pos_sapa_2026_source_meta_v1";

        // =========================================================================
        // 2. INISIALISASI & PENYIMPANAN DATA
        // =========================================================================
        function initApp() {
            const savedData = localStorage.getItem(STORAGE_KEY);
            const savedMeta = localStorage.getItem(SOURCE_META_KEY);

            if (savedData) {
                try {
                    allRecords = JSON.parse(savedData);
                    if (savedMeta) {
                        const meta = JSON.parse(savedMeta);
                        updateSourceBadge(meta.label, meta.isLive);
                    }
                } catch (e) {
                    allRecords = JSON.parse(JSON.stringify(SAMPLE_RECORDS));
                }
            } else {
                allRecords = JSON.parse(JSON.stringify(SAMPLE_RECORDS));
                updateSourceBadge(`Data Simulasi Bawaan (${allRecords.length} Laporan)`, false);
            }

            populateFilterDropdowns();
            applyGlobalFilters();
        }

        function persistRecords(sourceLabel, isLive = false) {
            localStorage.setItem(STORAGE_KEY, JSON.stringify(allRecords));
            localStorage.setItem(SOURCE_META_KEY, JSON.stringify({ label: sourceLabel, isLive }));
            updateSourceBadge(sourceLabel, isLive);
        }

        function updateSourceBadge(label, isLive) {
            document.getElementById('sync-source-text').textContent = label;
            const dot = document.getElementById('sync-dot');
            if (isLive) {
                dot.className = "w-2.5 h-2.5 rounded-full bg-amber-400 animate-ping";
            } else {
                dot.className = "w-2.5 h-2.5 rounded-full bg-emerald-400";
            }
        }

        // =========================================================================
        // 3. FILTER GLOBAL & PEMBARUAN TAMPILAN
        // =========================================================================
        function populateFilterDropdowns() {
            const kabSet = new Set();
            const lokSet = new Set();
            const perSet = new Set();

            allRecords.forEach(r => {
                if (r.kabKota) kabSet.add(r.kabKota.trim());
                if (r.lokasi) lokSet.add(r.lokasi.trim());
                if (r.periode) perSet.add(r.periode.trim());
            });

            fillSelectOptions('filter-kabkota', 'Semua Kabupaten / Kota', Array.from(kabSet).sort());
            fillSelectOptions('filter-lokasi', 'Semua Lokasi Pos SAPA', Array.from(lokSet).sort());
            fillSelectOptions('filter-periode', 'Semua Periode Laporan', Array.from(perSet));
        }

        function fillSelectOptions(selectId, defaultText, items) {
            const el = document.getElementById(selectId);
            const currentVal = el.value;
            el.innerHTML = `<option value="ALL">${defaultText}</option>`;
            items.forEach(item => {
                const opt = document.createElement('option');
                opt.value = item;
                opt.textContent = item;
                el.appendChild(opt);
            });
            if (items.includes(currentVal)) {
                el.value = currentVal;
            }
        }

        function applyGlobalFilters() {
            const kab = document.getElementById('filter-kabkota').value;
            const lok = document.getElementById('filter-lokasi').value;
            const per = document.getElementById('filter-periode').value;
            const sim = document.getElementById('filter-simfoni').value;

            filteredRecords = allRecords.filter(r => {
                if (kab !== 'ALL' && r.kabKota !== kab) return false;
                if (lok !== 'ALL' && r.lokasi !== lok) return false;
                if (per !== 'ALL' && r.periode !== per) return false;

                if (sim !== 'ALL') {
                    // Filter SIMFONI berlaku untuk data pengaduan & rincian kasus
                    if (r.kasus) {
                        if (r.kasus.sudahSimfoni !== sim) return false;
                    } else if (r.rekapPengaduan) {
                        const hasSimfoni = (r.rekapPengaduan.tercatatSimfoni || 0) > 0 ? 'Ya' : 'Belum';
                        if (hasSimfoni !== sim) return false;
                    } else {
                        return false;
                    }
                }
                return true;
            });

            renderAllSections();
        }

        function resetGlobalFilters() {
            document.getElementById('filter-kabkota').value = 'ALL';
            document.getElementById('filter-lokasi').value = 'ALL';
            document.getElementById('filter-periode').value = 'ALL';
            document.getElementById('filter-simfoni').value = 'ALL';
            applyGlobalFilters();
            showToast("Semua filter telah direset.", "info");
        }

        function switchTab(tabKey) {
            document.querySelectorAll('.tab-panel').forEach(el => el.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('active');
                btn.classList.add('text-slate-600');
            });

            document.getElementById(`tab-content-${tabKey}`).classList.remove('hidden');
            const activeBtn = document.getElementById(`tab-btn-${tabKey}`);
            activeBtn.classList.add('active');
            activeBtn.classList.remove('text-slate-600');
        }

        // =========================================================================
        // 4. KALKULASI KPI & RENDER SELURUH KOMPONEN
        // =========================================================================
        function renderAllSections() {
            const anakList = filteredRecords.filter(r => r.anak);
            const perempuanList = filteredRecords.filter(r => r.perempuan);
            const rekapList = filteredRecords.filter(r => r.rekapPengaduan);
            const kasusList = filteredRecords.filter(r => r.kasus);
            const medsosList = filteredRecords.filter(r => r.medsos);

            // Update Badge Hitungan pada Tab
            document.getElementById('badge-count-anak').textContent = anakList.length;
            document.getElementById('badge-count-perempuan').textContent = perempuanList.length;
            document.getElementById('badge-count-pengaduan').textContent = rekapList.length + kasusList.length;
            document.getElementById('badge-count-medsos').textContent = medsosList.length;

            // --- KALKULASI KPI 1: ANAK ---
            let anakTotal = 0, anakUsia0_5 = 0, anakUsia6_12 = 0, anakUsia13_17 = 0;
            let anakL = 0, anakP = 0, anakDis = 0, anakTL = 0;

            anakList.forEach(r => {
                const a = r.anak;
                const sumUsia = (Number(a.usia0_5) || 0) + (Number(a.usia6_12) || 0) + (Number(a.usia13_17) || 0);
                const sumGender = (Number(a.lakiLaki) || 0) + (Number(a.perempuan) || 0);
                anakTotal += Math.max(sumUsia, sumGender);
                anakUsia0_5 += Number(a.usia0_5) || 0;
                anakUsia6_12 += Number(a.usia6_12) || 0;
                anakUsia13_17 += Number(a.usia13_17) || 0;
                anakL += Number(a.lakiLaki) || 0;
                anakP += Number(a.perempuan) || 0;
                anakDis += Number(a.disabilitas) || 0;
                anakTL += Number(a.perluLayananLanjutan) || 0;
            });

            document.getElementById('kpi-anak-total').textContent = anakTotal.toLocaleString('id-ID');
            document.getElementById('kpi-anak-disabilitas').textContent = anakDis;
            document.getElementById('kpi-anak-tl').textContent = anakTL;

            // --- KALKULASI KPI 2: PEREMPUAN ---
            let prTotal = 0, prKK = 0, prHamil = 0, prLansia = 0, prDis = 0;
            perempuanList.forEach(r => {
                const p = r.perempuan;
                prTotal += Number(p.totalIkut) || 0;
                prKK += Number(p.kepalaKeluarga) || 0;
                prHamil += Number(p.hamilMenyusui) || 0;
                prLansia += Number(p.lansia) || 0;
                prDis += Number(p.disabilitas) || 0;
            });

            document.getElementById('kpi-perempuan-total').textContent = prTotal.toLocaleString('id-ID');
            document.getElementById('kpi-perempuan-kk').textContent = prKK;
            document.getElementById('kpi-perempuan-hamil').textContent = prHamil;
            document.getElementById('kpi-perempuan-rentan').textContent = prLansia + prDis;

            // --- KALKULASI KPI 3: PENGADUAN & SIMFONI ---
            let aduanMasuk = 0, aduanTL = 0, simfoniRekap = 0;
            rekapList.forEach(r => {
                aduanMasuk += Number(r.rekapPengaduan.aduanMasuk) || 0;
                aduanTL += Number(r.rekapPengaduan.sudahDitindaklanjuti) || 0;
                simfoniRekap += Number(r.rekapPengaduan.tercatatSimfoni) || 0;
            });

            // Jika rekap bulanan kosong tapi ada rincian kasus, gunakan rincian kasus sebagai cadangan hitungan
            const kasusSimfoniYa = kasusList.filter(r => r.kasus.sudahSimfoni === 'Ya').length;
            const simfoniPct = kasusList.length > 0
                ? Math.round((kasusSimfoniYa / kasusList.length) * 100)
                : (aduanMasuk > 0 ? Math.round((simfoniRekap / aduanMasuk) * 100) : 0);

            const displayAduanTotal = aduanMasuk > 0 ? aduanMasuk : kasusList.length;
            const displayAduanTL = aduanMasuk > 0 ? aduanTL : kasusList.filter(r => r.kasus.statusPenanganan !== '').length;
            const tlPct = displayAduanTotal > 0 ? Math.round((displayAduanTL / displayAduanTotal) * 100) : 0;

            document.getElementById('kpi-aduan-total').textContent = displayAduanTotal;
            document.getElementById('kpi-aduan-tl').textContent = displayAduanTL;
            document.getElementById('kpi-aduan-tl-pct').textContent = `${tlPct}% TL`;
            document.getElementById('kpi-simfoni-pct').textContent = `${simfoniPct}%`;

            // --- KALKULASI KPI 4: MEDSOS & DOKUMENTASI ---
            const dokCount = filteredRecords.filter(r => (r.dokumentasi || '').toLowerCase().includes('ada dokumentasi') && !(r.dokumentasi || '').toLowerCase().includes('tidak')).length;
            const dokPct = filteredRecords.length > 0 ? Math.round((dokCount / filteredRecords.length) * 100) : 0;

            document.getElementById('kpi-medsos-total').textContent = medsosList.length;
            document.getElementById('kpi-laporan-total').textContent = filteredRecords.length;
            document.getElementById('kpi-dok-pct').textContent = `${dokPct}%`;

            // Render detail masing-masing tab
            renderOverviewTab(filteredRecords, rekapList, anakList, kasusList);
            renderAnakTab(anakList, { anakUsia0_5, anakUsia6_12, anakUsia13_17, anakL, anakP, anakDis, anakTL });
            renderPerempuanTab(perempuanList, { prTotal, prKK, prHamil, prLansia, prDis });
            renderPengaduanTab(rekapList, kasusList);
            renderMedsosTab(medsosList);
        }

        function renderChart(canvasId, type, data, extraOptions = {}) {
            const ctx = document.getElementById(canvasId);
            if (!ctx) return;

            if (charts[canvasId]) {
                charts[canvasId].destroy();
            }

            charts[canvasId] = new Chart(ctx.getContext('2d'), {
                type: type,
                data: data,
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        tooltip: {
                            backgroundColor: '#0f172a',
                            padding: 10,
                            cornerRadius: 8
                        }
                    },
                    ...extraOptions
                }
            });
        }

        async function syncFromGoogleSheetUrl() {
            const url = document.getElementById('gsheet-csv-url').value.trim();
            if (!url) {
                showToast("Harap masukkan URL CSV publik Google Sheets terlebih dahulu.", "error");
                return;
            }

            const btn = document.getElementById('btn-fetch-csv');
            const origHTML = btn.innerHTML;
            btn.innerHTML = `<i class="fa-solid fa-spinner fa-spin"></i> Mengambil...`;
            btn.disabled = true;

            try {
                const response = await fetch(url);
                if (!response.ok) throw new Error("Gagal mengakses URL CSV.");
                const csvText = await response.text();
                parseAndLoadCSVText(csvText, "Google Sheets Live CSV");
                closeSyncModal();
            } catch (err) {
                console.error(err);
                showToast("Gagal menarik CSV dari URL. Pastikan sheet sudah 'Dipublikasikan ke web' sebagai CSV atau gunakan fitur Unggah File.", "error");
            } finally {
                btn.innerHTML = origHTML;
                btn.disabled = false;
            }
        }

        function handleFileUpload(event) {
            const file = event.target.files[0];
            if (!file) return;

            const fileName = file.name.toLowerCase();
            if (fileName.endsWith('.csv')) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    parseAndLoadCSVText(e.target.result, `File CSV: ${file.name}`);
                    closeSyncModal();
                };
                reader.readAsText(file);
            } else if (fileName.endsWith('.xlsx') || fileName.endsWith('.xls')) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    const data = new Uint8Array(e.target.result);
                    const workbook = XLSX.read(data, { type: 'array' });
                    const firstSheetName = workbook.SheetNames[0];
                    const csvText = XLSX.utils.sheet_to_csv(workbook.Sheets[firstSheetName]);
                    parseAndLoadCSVText(csvText, `File Excel: ${file.name}`);
                    closeSyncModal();
                };
                reader.readAsArrayBuffer(file);
            } else {
                showToast("Format file tidak didukung. Gunakan .csv atau .xlsx.", "error");
            }
        }

        // Pemetaan cerdas kolom Google Form (59 Pertanyaan) ke struktur dashboard
        function parseAndLoadCSVText(csvText, sourceLabel) {
            Papa.parse(csvText, {
                header: true,
                skipEmptyLines: true,
                complete: function(results) {
                    const rows = results.data;
                    if (!rows || rows.length === 0) {
                        showToast("File CSV kosong atau tidak memiliki baris data.", "error");
                        return;
                    }

                    const mapped = rows.map((row, idx) => mapGoogleFormRowToRecord(row, idx)).filter(Boolean);
                    if (mapped.length === 0) {
                        showToast("Tidak ditemukan kolom yang sesuai dengan format Google Form Pos SAPA.", "error");
                        return;
                    }

                    allRecords = mapped;
                    persistRecords(`${sourceLabel} (${allRecords.length} Baris)`, true);
                    populateFilterDropdowns();
                    applyGlobalFilters();
                    showToast(`Berhasil menyinkronkan ${allRecords.length} laporan dari ${sourceLabel}!`, "success");
                },
                error: function(err) {
                    console.error(err);
                    showToast("Terjadi kesalahan saat membaca file CSV.", "error");
                }
            });
        }

        function findColValue(row, keywords) {
            const keys = Object.keys(row);
            for (const kw of keywords) {
                const foundKey = keys.find(k => k.toLowerCase().includes(kw.toLowerCase()));
                if (foundKey && row[foundKey] !== undefined && String(row[foundKey]).trim() !== '') {
                    return String(row[foundKey]).trim();
                }
            }
            return '';
        }

            document.querySelectorAll('.form-branch').forEach(el => el.classList.add('hidden'));

            if (val.includes('Aktivitas Anak')) {
                document.getElementById('branch-anak').classList.remove('hidden');
            } else if (val.includes('Aktivitas Perempuan')) {
                document.getElementById('branch-perempuan').classList.remove('hidden');
            } else if (val.includes('Rekapitulasi Layanan Pengaduan')) {
                document.getElementById('branch-rekap').classList.remove('hidden');
            } else if (val.includes('Rincian Tindak Lanjut')) {
                document.getElementById('branch-kasus').classList.remove('hidden');
            } else if (val.includes('Publikasi Media Sosial')) {
                document.getElementById('branch-medsos').classList.remove('hidden');
            }
        }

        function showToast(message, type = 'info') {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            
            let bg = 'bg-slate-900 text-white border-slate-700';
            let icon = 'fa-circle-info text-amber-400';

            if (type === 'success') {
                bg = 'bg-emerald-800 text-white border-emerald-600';
                icon = 'fa-circle-check text-emerald-300';
            } else if (type === 'error') {
                bg = 'bg-rose-800 text-white border-rose-600';
                icon = 'fa-circle-exclamation text-rose-300';
            }

            toast.className = `${bg} border px-4 py-3 rounded-xl shadow-xl flex items-center gap-3 text-xs sm:text-sm font-medium transform transition-all duration-300 translate-y-2 opacity-0 pointer-events-auto max-w-md`;
            toast.innerHTML = `<i class="fa-solid ${icon} text-base shrink-0"></i><span>${message}</span>`;

            container.appendChild(toast);
            requestAnimationFrame(() => toast.classList.remove('translate-y-2', 'opacity-0'));

            setTimeout(() => {
                toast.classList.add('translate-y-2', 'opacity-0');
                setTimeout(() => toast.remove(), 300);
            }, 3500);
        }
